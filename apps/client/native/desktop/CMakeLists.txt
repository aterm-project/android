cmake_minimum_required(VERSION 3.28)
project(atermius_rdp LANGUAGES C CXX)
include(FetchContent)
# Build the protocol engine into the application. No local daemon or CLI viewer.
# Redirection channels use only capabilities already present in Windows
# (Win32 files, winmm, spooler, winscard, Media Foundation). No driver is
# installed or replaced. Generic USB (libusb) is opt-in and only reaches
# devices already bound to WinUSB.
if(WIN32)
  set(_windows_default ON)
else()
  set(_windows_default OFF)
endif()
option(ATERMIUS_RDP_DEVICES "Folder (drive) redirection" ON)
option(ATERMIUS_RDP_PRINTER "Printer redirection (Windows spooler backend)" ${_windows_default})
option(ATERMIUS_RDP_SMARTCARD "Smart card redirection (system WinSCard)" ${_windows_default})
option(ATERMIUS_RDP_AUDIO "Audio playback/capture (winmm)" ${_windows_default})
option(ATERMIUS_RDP_CAMERA "Camera redirection (Media Foundation HAL)" ${_windows_default})
option(ATERMIUS_RDP_USB "Generic USB redirection via libusb (WinUSB-bound devices only)" OFF)
if(ATERMIUS_RDP_CAMERA AND NOT WIN32)
  message(FATAL_ERROR "The Atermius camera HAL uses Media Foundation (Windows only)")
endif()
# libusb is LGPL-2.1: on Windows link it dynamically from a separate vcpkg
# x64-windows (DLL) prefix and ship libusb-1.0.dll beside the engine. No USB
# driver is installed; only devices already bound to WinUSB can be opened.
set(ATERMIUS_LIBUSB_ROOT "" CACHE PATH "Dynamic libusb prefix, e.g. <vcpkg>/installed/x64-windows")
if(ATERMIUS_RDP_USB AND WIN32)
  if(NOT ATERMIUS_LIBUSB_ROOT)
    message(FATAL_ERROR "ATERMIUS_RDP_USB=ON needs ATERMIUS_LIBUSB_ROOT (vcpkg install libusb:x64-windows)")
  endif()
  set(_libusb_include "${ATERMIUS_LIBUSB_ROOT}/include/libusb-1.0")
  set(_libusb_import "${ATERMIUS_LIBUSB_ROOT}/lib/libusb-1.0.lib")
  set(ATERMIUS_LIBUSB_DLL "${ATERMIUS_LIBUSB_ROOT}/bin/libusb-1.0.dll")
  foreach(_required "${_libusb_include}/libusb.h" "${_libusb_import}" "${ATERMIUS_LIBUSB_DLL}")
    if(NOT EXISTS "${_required}")
      message(FATAL_ERROR "Dynamic libusb file missing: ${_required}")
    endif()
  endforeach()
  # Pre-seed FreeRDP's Findlibusb-1.0 so it cannot pick a static copy.
  set(LIBUSB_1_INCLUDE_DIRS "${_libusb_include}" CACHE PATH "" FORCE)
  set(LIBUSB_1_LIBRARIES "${_libusb_import}" CACHE FILEPATH "" FORCE)
endif()

foreach(channel RDPDR RDPSND AUDIN AINPUT RAIL RDPGFX RDPEI GEOMETRY VIDEO ENCOMSP REMDESK ECHO URBDRC TSMF SMARTCARD PRINTER DRIVE SERIAL PARALLEL LOCATION RDP2TCP RDPECAM)
  set(CHANNEL_${channel} OFF CACHE BOOL "" FORCE)
endforeach()
if(ATERMIUS_RDP_DEVICES OR ATERMIUS_RDP_PRINTER OR ATERMIUS_RDP_SMARTCARD)
  set(CHANNEL_RDPDR ON CACHE BOOL "" FORCE)
endif()

set(CHANNEL_PRINTER ${ATERMIUS_RDP_PRINTER} CACHE BOOL "" FORCE)
set(CHANNEL_SMARTCARD ${ATERMIUS_RDP_SMARTCARD} CACHE BOOL "" FORCE)
# FreeRDP always pairs rdpdr with an rdpsnd channel (sys:fake when audio is
# off) because Windows servers expect it; the channel must therefore be built
# whenever device redirection is. Playback still requires ATERMIUS_RDP_AUDIO.
if(ATERMIUS_RDP_AUDIO OR CHANNEL_RDPDR)
  set(CHANNEL_RDPSND ON CACHE BOOL "" FORCE)
else()
  set(CHANNEL_RDPSND OFF CACHE BOOL "" FORCE)
endif()
set(CHANNEL_AUDIN ${ATERMIUS_RDP_AUDIO} CACHE BOOL "" FORCE)
set(CHANNEL_RDPECAM ${ATERMIUS_RDP_CAMERA} CACHE BOOL "" FORCE)
set(CHANNEL_URBDRC ${ATERMIUS_RDP_USB} CACHE BOOL "" FORCE)
set(WITH_CHANNELS ON CACHE BOOL "" FORCE)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
FetchContent_Declare(freerdp
  URL https://github.com/FreeRDP/FreeRDP/releases/download/3.33.0/freerdp-3.33.0.tar.gz
  URL_HASH SHA256=80cbcf17a7de00909f2be805c7d93764596cd712046addc18c30594137c0b879
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  # Applied once after extraction, before FreeRDP is configured.
  PATCH_COMMAND ${CMAKE_COMMAND} -DFREERDP_SOURCE=<SOURCE_DIR>
    -P ${CMAKE_CURRENT_SOURCE_DIR}/freerdp_patches.cmake)
if(FETCHCONTENT_SOURCE_DIR_FREERDP)
  # A pre-extracted source bypasses PATCH_COMMAND; patch it the same way.
  execute_process(COMMAND ${CMAKE_COMMAND} -DFREERDP_SOURCE=${FETCHCONTENT_SOURCE_DIR_FREERDP}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/freerdp_patches.cmake RESULT_VARIABLE _patched)
  if(NOT _patched EQUAL 0)
    message(FATAL_ERROR "FreeRDP source patches failed")
  endif()
endif()
FetchContent_MakeAvailable(freerdp)
include(${CMAKE_CURRENT_SOURCE_DIR}/bitmap_padding.cmake)
find_package(OpenSSL REQUIRED)
set(_sources atermius_rdp.c)
if(ATERMIUS_RDP_CAMERA)
  list(APPEND _sources camera_mf.c)
endif()
add_library(atermius_rdp SHARED ${_sources})
if(WIN32)
  target_compile_definitions(atermius_rdp PRIVATE FREERDP_EXPORTS)
endif()
foreach(_feature DEVICES PRINTER SMARTCARD AUDIO CAMERA USB)
  if(ATERMIUS_RDP_${_feature})
    target_compile_definitions(atermius_rdp PRIVATE ATERMIUS_RDP_${_feature})
  endif()
endforeach()
target_link_libraries(atermius_rdp PRIVATE freerdp-client freerdp winpr OpenSSL::Crypto)
if(ATERMIUS_RDP_CAMERA)
  target_link_libraries(atermius_rdp PRIVATE mfplat mfreadwrite mf mfuuid ole32)
  target_include_directories(atermius_rdp PRIVATE ${freerdp_SOURCE_DIR}/channels/rdpecam/client)
endif()
target_include_directories(atermius_rdp PRIVATE
  ${freerdp_SOURCE_DIR}/include ${freerdp_BINARY_DIR}/include
  ${freerdp_SOURCE_DIR}/winpr/include ${freerdp_BINARY_DIR}/winpr/include)
set_target_properties(atermius_rdp PROPERTIES C_STANDARD 11)
if(ATERMIUS_RDP_USB AND WIN32)
  add_custom_command(TARGET atermius_rdp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ATERMIUS_LIBUSB_DLL}" "$<TARGET_FILE_DIR:atermius_rdp>")
  install(FILES "${ATERMIUS_LIBUSB_DLL}" DESTINATION .)
endif()
install(TARGETS atermius_rdp RUNTIME DESTINATION . LIBRARY DESTINATION .)
